from pathlib import Path
import hashlib
import json
import re
import subprocess
from urllib.parse import unquote, urlparse

ROOT = Path('/home/balazs/.t3/worktrees/DynaDocs/t3-a2f64ef5-eadbcb161c93938a')
BASE = 'b995a1927c9ad29a78f35fc9ee1d5f82ee3b358c'
EVIDENCE = ROOT / 'dydo/project/changelog/2026/2026-11-18/dyd-1186'

def git(*args):
    return subprocess.check_output(['git', *args], cwd=ROOT, text=True)

def check(condition, message):
    assert condition, message
    print('OK:', message)

(EVIDENCE / 'source-link-probe.txt').write_text(Path(__file__).read_text())
check(git('branch', '--show-current').strip() == 'DYD-1186-doc-rulings', 'ticket branch')
check(git('rev-parse', '--show-toplevel').strip() == str(ROOT), 'isolated worktree root')
check(subprocess.run(['git', 'merge-base', '--is-ancestor', BASE, 'HEAD'], cwd=ROOT).returncode == 0,
      'posted base is an ancestor')

changed = git('diff', '--name-only', BASE).splitlines()
untracked = git('ls-files', '--others', '--exclude-standard').splitlines()
all_changed = sorted(set(changed + untracked))
check(not any(p.startswith(('skills/', 'claims/')) for p in all_changed), 'no excluded-path edits')
check(all(p == 'README.md' or p.startswith(('dydo/', 'Scaffold/dydo/')) for p in all_changed),
      'every edited path is owned documentation or its evidence')
check(all(Path(p).suffix in {'.md', '.txt', '.log', '.json'} for p in all_changed),
      'no source/runtime code or tests edited')
(EVIDENCE / 'affected-files.txt').write_text('\n'.join(all_changed) + '\n')

deleted = git('diff', '--name-only', '--diff-filter=D', BASE).splitlines()
expected = (EVIDENCE / 'deleted-plans.txt').read_text().splitlines()
check(len(deleted) == 11 and sorted(deleted) == sorted(expected), 'exact eleven plan files deleted')
check([p.name for p in (ROOT / 'dydo/project/plans').iterdir()] == ['_plans.md'],
      'plans index retained alone')

envelopes = [json.loads(line) for line in (EVIDENCE / 'linear-snapshot.txt').read_text().splitlines() if line.strip()]
decision = json.loads(envelopes[1]['value']['content'][0]['text'])['comments'][0]['body']
statement = re.search(r'- Statement: "(.*?)"', decision).group(1)
description = re.search(r'- Description, verbatim: "(.*?)"', decision).group(1)
design = (ROOT / 'dydo/law/dydo/design.md').read_text()
section = design.split('## dydo-design-shared-language\n\n', 1)[1].split('\n## ', 1)[0]
check(section.startswith(statement + '\n\ndescription: ' + description + '\n\n'),
      'principle statement and description match governing comment verbatim')
check('cites: [dydo-aligned-intent](./axioms.md#dydo-aligned-intent)' in section
      and '#comment-bdb6b643-f33d-4bd0-b801-c33566d63bca)' in section,
      'principle citation and origin')
check('## dydo-aligned-intent' in (ROOT / 'dydo/law/dydo/axioms.md').read_text(),
      'cited axiom anchor exists')

live = [p for p in git('ls-files').splitlines()
        if (p == 'README.md' or p.startswith(('dydo/', 'Scaffold/dydo/')))
        and '/project/' not in p and p.endswith('.md')]
for path in live:
    content = (ROOT / path).read_text()
    check(not re.search(r'\b(?:Project[ -]plan|architecture[ -]plans?|repository plans?|reviewed plans)\b',
                        content, re.I), 'no abolished plan kind in live doc: ' + path)

nesting = ['dydo/guides/working-tree-standard.md', 'Scaffold/dydo/guides/working-tree-standard.md',
           'dydo/understand/control-flow.md', 'dydo/reference/linear-workspace-standard.md',
           'Scaffold/dydo/reference/linear-workspace-standard.md']
for path in nesting:
    content = (ROOT / path).read_text()
    check(not re.search(r'lanes have one level|one level deep|sibling lanes|depth \| one level', content, re.I),
          'no contradictory one-level or sibling-only rule: ' + path)
    for term in ['no hard cap', 'officer above', 'too big, re-cut']:
        check(term in content.lower(), term + ': ' + path)
for path in nesting[:3]:
    content = (ROOT / path).read_text().lower()
    for term in ['self-similar', 'admiral → captain → sub-captain → second-order sub-captain → crew',
                 'without a separate pilot']:
        check(term in content, term + ': ' + path)

def changed_lines(path):
    return [line for line in git('diff', '--unified=0', BASE, '--', path).splitlines()
            if line.startswith(('+', '-')) and not line.startswith(('+++', '---'))]

for path in ['dydo/guides/working-tree-standard.md', 'dydo/reference/dydo-glossary.md',
             'dydo/reference/about-dynadocs.md', 'dydo/reference/feature-spec-convention.md',
             'dydo/reference/linear-workspace-standard.md']:
    check(changed_lines(path) == changed_lines('Scaffold/' + path),
          'canonical twin receives identical amended lines: ' + path)

targets = {(ROOT / p).resolve() for p in expected}
tracked = git('ls-files').splitlines()
excluded = []
permalinks = 0
for name in tracked:
    path = ROOT / name
    if not path.is_file() or path.suffix not in {'.md', '.json', '.txt', '.yaml', '.yml', '.html'}:
        continue
    content = path.read_text()
    # Inline links and reference-style definitions; URI paths are checked below.
    links = re.findall(r'\]\(<?([^\s)>]+)>?(?:\s+[^)]*)?\)', content)
    links += re.findall(r'^\s*\[[^\]]+\]:\s*<?([^\s>]+)>?', content, re.M)
    for target in links:
        if urlparse(target).scheme or target.startswith('#'):
            continue
        if (path.parent / unquote(target.split('#', 1)[0])).resolve() in targets:
            if name.startswith(('skills/', 'claims/')):
                excluded.append(name + ': ' + target)
            else:
                raise AssertionError('broken inbound deleted-plan link: ' + name + ': ' + target)
    # Check all GitHub plan URLs, including raw JSON URLs, not historical path manifests.
    for match in re.finditer(r'https://github\.com/[^/]+/dydo/blob/([^/]+)/(dydo/project/plans/[^\s)\]<>"\\]+)', content):
        ref, target = match.groups()
        target = target.split('#', 1)[0]
        if target not in expected:
            continue
        if ref in {'master', 'main'}:
            raise AssertionError('branch-following inbound deleted-plan URL: ' + name)
        result = subprocess.run(['git', 'cat-file', '-e', ref + ':' + target], cwd=ROOT,
                                stdout=subprocess.DEVNULL, stderr=subprocess.DEVNULL)
        check(result.returncode == 0, 'historical Git permalink object: ' + name + ' -> ' + ref + ':' + target)
        permalinks += 1
check(not excluded, 'no excluded skill/claim deleted-plan inbound link sites')
print('Validated historical plan permalink occurrences:', permalinks)

# Historical records changed only by repairing inbound links, preserving all other text.
def normalize_history(text):
    text = re.sub(r'https://github\.com/[^/]+/dydo/blob/[^/]+/dydo/project/plans/([^\s)\]<>"\\]+)', r'PLAN/\1', text)
    return re.sub(r'\]\((?:\.\./)+plans/([^\s)]+)\)', r'](PLAN/\1)', text)

        check(normalize_history(git('show', BASE + ':' + name)) == normalize_history((ROOT / name).read_text()),
              'historical meaning preserved; link-only diff: ' + name)
