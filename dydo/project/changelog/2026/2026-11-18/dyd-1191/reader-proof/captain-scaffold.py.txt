from pathlib import Path
import shutil
import subprocess
import tempfile
root=Path('/home/balazs/.t3/worktrees/DynaDocs/t3-be8f42a8-69c8583eb5e211b4')
with tempfile.TemporaryDirectory(prefix='dyd1191-captain-scaffold-') as tmp:
    scratch=Path(tmp)
    shutil.copytree(root/'Scaffold/dydo',scratch/'dydo')
    shutil.copyfile(root/'dydo.json',scratch/'dydo.json')
    for verb in ['fix','check']:
        result=subprocess.run(['dydo',verb],cwd=scratch)
        assert result.returncode==0
    for original in (root/'Scaffold/dydo').rglob('*'):
        if original.is_file(): assert original.read_bytes()==(scratch/'dydo'/original.relative_to(root/'Scaffold/dydo')).read_bytes()
print('PASS Scaffold check; candidate bytes unchanged; owned temporary tree removed')
