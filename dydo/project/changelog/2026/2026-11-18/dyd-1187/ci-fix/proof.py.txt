from pathlib import Path
import ast
import hashlib
import json
import subprocess
import sys
import unittest

root = Path.cwd()
out = Path('/tmp/dyd-1187-ruling-proof')
out.mkdir(exist_ok=True)
sys.path[:0] = [str(root / 'DynaDocs.Tests/coverage'), str(root / 'DynaDocs.Tests/coverage/tests')]
from associations import validate_associations
report = json.loads(Path('/tmp/dyd-1187-ci-python/adapters/python-static.json').read_text())
inventory_path = next(Path('/tmp/dyd-1187-ci-python/assurance').glob('*/inventory.json'))
inventory = json.loads(inventory_path.read_text())
sources = inventory['sources']
manifest = json.loads(Path('DynaDocs.Tests/coverage/test-associations.json').read_text())
association_findings = validate_associations(sources, manifest)
assert association_findings == [], association_findings
(out / 'associations.json').write_text(json.dumps({'originalFindings': report['findings'], 'currentFindings': association_findings, 'inventory': str(inventory_path)}, indent=2) + '\n')
print('Native association validator:', len(association_findings), 'findings', flush=True)
import test_testing_facade
suite = unittest.TestSuite(cls('test_owned_policy_docs_replace_tiers_with_the_dr048_gate_set')
                          for cls in [test_testing_facade.TestingFacadeTests, test_testing_facade.PortableTestingFacadeTests])
with (out / 'python-doc-tests.log').open('w') as log:
    result = unittest.TextTestRunner(stream=log, verbosity=2).run(suite)
assert result.wasSuccessful()
print('Python doc cases:', result.testsRun, 'passed', flush=True)
recipe = Path('dydo/project/changelog/2026/2026-11-18/dyd-1187/proof.py.txt')
light = next(ast.literal_eval(node.value) for node in ast.parse(recipe.read_text()).body
             if isinstance(node, ast.Assign) and any(isinstance(t, ast.Name) and t.id == 'LIGHT' for t in node.targets))
selection = light + '|FullyQualifiedName~OneCanonicalSkillTreeReachesEverySupportedHostFeature'
commands = []
for name, extra in [('discovery', ['--list-tests']), ('light', ['--logger', 'trx;LogFileName=light.trx', '--results-directory', str(out / 'light')])]:
    argv = ['dotnet', 'test', 'DynaDocs.Tests/DynaDocs.Tests.csproj', '--no-restore', '--filter', selection, '--verbosity', 'quiet', *extra]
    with (out / (name + '.log')).open('w') as log:
        process = subprocess.run(argv, stdout=log, stderr=subprocess.STDOUT)
    commands.append({'name': name, 'argv': argv, 'exit': process.returncode})
    (out / 'commands.json').write_text(json.dumps(commands, indent=2) + '\n')
    print(name, process.returncode, out / (name + '.log'), flush=True)
    if name == 'discovery' and process.returncode:
        raise SystemExit(process.returncode)
print('Expected Light exit 1 only for AllOptions_AppearInReferenceDoc (--base), pending DYD-1191.', flush=True)
