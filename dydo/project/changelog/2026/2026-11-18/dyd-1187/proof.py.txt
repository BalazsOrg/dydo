"""Run from the DYD-1187 worktree root; raw red exit 1 is expected. No checkout/branch is created."""
from pathlib import Path
import subprocess
import sys

BASE = 'b995a1927c9ad29a78f35fc9ee1d5f82ee3b358c'
PRODUCTION = ['Commands/CheckCommand.cs', 'Commands/CheckDocValidator.cs', 'Commands/PrReadyCommand.cs',
              'Services/ExplainsCheck.cs', 'Services/ExplainsDeclaration.cs']
LIGHT = 'FullyQualifiedName~ExplainsCommandTests|FullyQualifiedName~CheckCommandTests|FullyQualifiedName~CheckDocValidatorTests|FullyQualifiedName~PrReadyCommandTests|FullyQualifiedName~GuardPrReadyTests|FullyQualifiedName~GuardPrReadyBodyFileTests|FullyQualifiedName~DocScannerTests|FullyQualifiedName~Frontmatter|FullyQualifiedName~CommandDocConsistencyTests|FullyQualifiedName~InitCheckIntegrationTests|FullyQualifiedName~InitCommandTests|FullyQualifiedName~ScaffoldTreeTests|FullyQualifiedName~CanonicalSkillAssertionTests|FullyQualifiedName~CommandSmokeTests|FullyQualifiedName~HelpCommandTests|FullyQualifiedName~CheckAssertionTests|FullyQualifiedName~OptionalSummaries'
action = sys.argv[1]
if action not in ('red', 'green', 'discover', 'light'):
    raise SystemExit('Expected red, green, discover or light')
saved = {}
try:
    if action == 'red':
        for path in PRODUCTION:
            if subprocess.run(['git', 'diff', 'HEAD', '--quiet', '--', path]).returncode:
                raise SystemExit('Refusing pre-existing edits in ' + path)
        saved = {path: Path(path).read_bytes() for path in PRODUCTION}
        for path in PRODUCTION:
            old = subprocess.run(['git', 'show', BASE + ':' + path], capture_output=True)
            if old.returncode == 0:
                Path(path).write_bytes(old.stdout)
            else:
                Path(path).unlink()
    arguments = ['dotnet', 'test', 'DynaDocs.Tests/DynaDocs.Tests.csproj', '--no-restore', '--filter',
                 LIGHT if action in ('light', 'discover') else 'FullyQualifiedName~ExplainsCommandTests',
                 '--verbosity', 'quiet']
    if action == 'discover':
        arguments += ['--list-tests']
    prefix = ['flock', '/tmp/dydo-full-gate.lock', 'env', 'MSBUILDDISABLENODEREUSE=1',
              'DOTNET_CLI_USE_MSBUILD_SERVER=0', 'UseSharedCompilation=false',
              'systemd-run', '--user', '--scope']
    slices = [part for part in Path('/proc/self/cgroup').read_text().split('/') if part.endswith('.slice')]
    if slices:
        prefix += ['--slice=' + slices[-1]]
    prefix += ['-p', 'MemoryMax=8G', '-p', 'MemorySwapMax=0', '-p', 'OOMPolicy=continue', 'timeout', '3600']
    log = Path('/tmp/dyd-1187-rerun-' + action + '.log')
    with log.open('wb') as output:
        result = subprocess.run(prefix + arguments, stdout=output, stderr=subprocess.STDOUT)
    print('Command:', prefix + arguments)
    print('Exit:', result.returncode, 'Raw:', log)
finally:
    for path, content in saved.items():
        Path(path).write_bytes(content)
raise SystemExit(result.returncode)
