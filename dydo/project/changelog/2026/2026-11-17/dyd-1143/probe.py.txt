from pathlib import Path
import difflib
import hashlib
import json
import subprocess
import sys
import xml.etree.ElementTree as ET
from datetime import datetime, timezone

mode, label = sys.argv[1:]
source = Path('Services/SafeDeleteGitHub.cs')
original = source.read_bytes()
text = original.decode()
guard = '            if (target.Metadata == null || reference.ValueKind != JsonValueKind.Null) throw new InvalidDataException("Remote ref recreated or own-deletion receipt is invalid");\n'
assert text.count(guard) == 1
if mode == 'api-null':
    replacement = guard.replace(' || reference.ValueKind != JsonValueKind.Null', '')
    failed, passed = 1, 4
    expected = ['AfterOwnDeletion_ApiRefSurvivesWhileGitIsAbsent']
elif mode == 'receipt-metadata':
    replacement = guard.replace('target.Metadata == null || ', '')
    failed, passed = 1, 4
    expected = ['context: "missing metadata"']
elif mode == 'both-checks':
    replacement = ''
    failed, passed = 2, 3
    expected = ['AfterOwnDeletion_ApiRefSurvivesWhileGitIsAbsent', 'context: "missing metadata"']
else:
    raise ValueError(mode)
mutated = text.replace(guard, replacement)
run = Path('/tmp/dyd1143-runs') / label
selection = 'FullyQualifiedName~AfterOwnDeletion_ApiRefSurvivesWhileGitIsAbsent|FullyQualifiedName~OwnDeletionReceipt_MissingOrUnownedContextCannotAuthorizeApiAbsence|FullyQualifiedName~PublicFork_LeasedOwnDeletionAuthorizesRemainingLocalSteps'
try:
    source.write_bytes(mutated.encode())
    result = subprocess.run(['python3', '/tmp/dyd1143-run.py', label, 'probe', 'dotnet', 'test',
        'DynaDocs.Tests/DynaDocs.Tests.csproj', '--no-restore', '--filter', selection, '--logger', 'trx',
        '--results-directory', str(run / 'results')], check=False)
    completed = datetime.now(timezone.utc).isoformat()
    (run / 'mutation.patch').write_text(''.join(difflib.unified_diff(text.splitlines(True), mutated.splitlines(True), fromfile=str(source), tofile=str(source))))
    assert result.returncode == 1, result.returncode
    trx, = (run / 'results').glob('*.trx')
    ns = {'t': 'http://microsoft.com/schemas/VisualStudio/TeamTest/2010'}
    root = ET.parse(trx)
    counters = root.find('.//t:Counters', ns).attrib
    assert int(counters['failed']) == failed, counters
    assert int(counters['passed']) == passed, counters
    assert int(counters['notExecuted']) == 0, counters
    failures = [row for row in root.findall('.//t:UnitTestResult', ns) if row.attrib['outcome'] == 'Failed']
    names = [row.attrib['testName'] for row in failures]
    for name in expected:
        assert any(name in found for found in names), names
    proof = {'mode': mode, 'counters': counters, 'failed_tests': names, 'child_completed': completed,
             'original_sha256': hashlib.sha256(original).hexdigest(), 'mutant_sha256': hashlib.sha256(source.read_bytes()).hexdigest()}
    (run / 'proof.json').write_text(json.dumps(proof, indent=2) + '\n')
    print('Confirmed intended RED:', json.dumps(proof))
finally:
    # subprocess.run has completed before the source bytes are restored.
    source.write_bytes(original)
    if run.exists():
        (run / 'restoration.json').write_text(json.dumps({'restored': datetime.now(timezone.utc).isoformat(),
            'sha256': hashlib.sha256(source.read_bytes()).hexdigest(), 'matches_original': source.read_bytes() == original}, indent=2) + '\n')
    print('Restored exact production bytes after child completion.', flush=True)
