from pathlib import Path
import gzip
import hashlib
import json
import re
import shutil
import subprocess
import tarfile
import xml.etree.ElementTree as ET

root = Path.cwd()
out = root / 'dydo/project/changelog/2026/2026-11-17/dyd-1142/windows-path-fix'
source = 'DynaDocs.Tests/Integration/SafeDeleteGitHubTests.cs'
head = '3fcbcd689a5294ece2fb4270037f0df9cda80260'
assert subprocess.check_output(['git', 'rev-parse', 'HEAD'], text=True).strip() == head
digest = hashlib.sha256((root / source).read_bytes()).hexdigest()
stage = Path('/tmp/dyd1142-windows-bundle-02')
stage.mkdir(exist_ok=False)
summary = {'baseline': head, 'tested_source': source, 'tested_source_sha256': digest,
           'results': {}, 'native_windows_green': 'OWED to hosted exact candidate CI',
           'mutation': 'UNAVAILABLE Pending DYD-103 (inherited; not run or passed here)',
           'local_statechart': 'unmet (inherited; not passed here)'}
labels = ['pins', 'pins-elevated', 'discovery', 'focused', 'runtime-source', 'light', 'static', 'docs']
ns = {'t': 'http://microsoft.com/schemas/VisualStudio/TeamTest/2010'}
for label in labels:
    run = Path('/tmp/dyd1142-runs') / ('windows-fix-' + label)
    receipt = json.loads((run / 'receipt.json').read_text())
    assert receipt['head'] == head
    assert receipt['files'][source] == digest
    assert receipt['exit'] == (1 if label == 'pins' else 0), (label, receipt['exit'])
    row = {'exit': receipt['exit'], 'argv': receipt['argv'], 'raw': 'runs/' + run.name + '/raw.log'}
    if label in ('focused', 'light'):
        trx, = (run / 'results').glob('*.trx')
        tree = ET.parse(trx)
        counters = tree.find('.//t:Counters', ns).attrib
        expected = 42 if label == 'focused' else 390
        assert int(counters['total']) == int(counters['passed']) == expected
        assert int(counters['failed']) == int(counters['notExecuted']) == 0
        results = tree.findall('.//t:UnitTestResult', ns)
        assert len(results) == expected and all(result.attrib['outcome'] == 'Passed' for result in results)
        row['trx_counters'] = counters
    if label == 'discovery':
        lines = (run / 'raw.log').read_text().splitlines()
        discovered = [line.strip() for line in lines[lines.index('The following Tests are available:') + 1:] if line.strip()]
        assert len(discovered) == 390, len(discovered)
        for group in ('SafeDeleteGitHubTests.InitiallyAbsent', 'CommandDocConsistencyTests.', 'Archive_IsRegistered'):
            assert any(group in line for line in discovered), group
        row['discovered'] = len(discovered)
    summary['results'][label] = row
    shutil.copytree(run, stage / 'runs' / run.name)

raw = gzip.decompress((out / 'native-windows-red.log.gz').read_bytes())
red = raw.decode('utf-8-sig')
assert 'Failed:    27, Passed:   308, Skipped:     0, Total:   335' in red
assert red.count('SafeDeleteGitHubTests.AssertInitiallyAbsentLocalIntact(') == 27
summary['native_windows_red'] = {'exit': 1, 'failed': 27, 'passed': 308, 'skipped': 0,
    'sha256': hashlib.sha256(raw).hexdigest(), 'provenance': 'native-windows-red-provenance.txt',
    'candidate': head, 'executed_merge': '5d269fde', 'observed': 'supplied hosted log inspected; no local Windows run'}
services = subprocess.check_output(['git', 'ls-files', 'Services/SafeDelete*.cs'], text=True).splitlines()
unchanged = {}
for name in services:
    baseline = subprocess.check_output(['git', 'show', head + ':' + name])
    assert baseline == (root / name).read_bytes(), name
    unchanged[name] = hashlib.sha256(baseline).hexdigest()
summary['unchanged_production_sha256'] = unchanged
changed = subprocess.check_output(['git', 'diff', '--name-only', head], text=True).splitlines()
assert changed == [source], changed
(out / 'assertion.patch').write_bytes(subprocess.check_output(['git', 'diff', head, '--', source]))
assert subprocess.check_output(['git', 'rev-list', '--count', 'd97c1b4f..HEAD'], text=True).strip() == '6'
summary['six_prior_commits_preserved'] = True
shutil.copytree('/tmp/dyd1142-windows-runtime-source', stage / 'runtime-source')
static_log = (Path('/tmp/dyd1142-runs/windows-fix-static/raw.log')).read_text()
paths = re.findall(r'(?:/[^\s]+)?DynaDocs\.Tests/coverage/results/run-[^\s]+', static_log)
static_root, = {Path(path.rstrip('.,:')).parent if Path(path.rstrip('.,:')).suffix == '.json' else Path(path.rstrip('.,:')) for path in paths}
assert (static_root / 'result.json').is_file(), static_root
static_result = json.loads((static_root / 'result.json').read_text())
summary['static_result'] = static_result
shutil.copytree(static_root, stage / 'static-artifacts')
adapter_path = root / 'DynaDocs.Tests/coverage/results/adapters/dotnet-static.json'
adapter = json.loads(adapter_path.read_text())
assert adapter['exitCode'] == 0 and adapter['measurementComplete']
assert adapter['findings'] == adapter['gaps'] == []
summary['static_adapter'] = {key: adapter[key] for key in ('candidate', 'exitCode', 'measurementComplete', 'findings', 'gaps')}
metrics = []
def inspect(value):
    if isinstance(value, dict):
        if 'policy_cc' in value and (value.get('member') == 'AssertInitiallyAbsentLocalIntact'
                or value.get('type', '').endswith('.AssertInitiallyAbsentLocalIntact')):
            metrics.append(value)
        for child in value.values():
            inspect(child)
    elif isinstance(value, list):
        for child in value:
            inspect(child)
inspect(adapter)
assert metrics and all(row['cognitive'] <= 20 and row['policy_cc'] ** 2 + row['cognitive'] <= 20 for row in metrics)
summary['changed_helper_metrics'] = metrics
shutil.copyfile(adapter_path, stage / 'static-artifacts/dotnet-static.json')
assurance = root / 'DynaDocs.Tests/coverage/results' / Path(adapter['inventory']['path']).parent
shutil.copytree(assurance, stage / 'static-artifacts/assurance')
summary['runtime_source_provenance'] = json.loads((stage / 'runtime-source/provenance.json').read_text())
(out / 'run.py.txt').write_bytes(Path('/tmp/dyd1142-windows-run.py').read_bytes())
(out / 'static-prerequisites.py.txt').write_bytes(Path('/tmp/dyd1142-windows-prerequisites.py').read_bytes())
(out / 'runtime-source.py.txt').write_bytes(Path('/tmp/dyd1142-windows-runtime-source.py').read_bytes())
(out / 'summary.json').write_text(json.dumps(summary, indent=2) + '\n')
with tarfile.open(out / 'raw-evidence.tar.gz', 'w:gz') as bundle:
    for item in sorted(stage.iterdir()):
        bundle.add(item, arcname=item.name)
archive = out / 'raw-evidence.tar.gz'
(out / 'raw-evidence.tar.gz.sha256').write_text(hashlib.sha256(archive.read_bytes()).hexdigest() + '  raw-evidence.tar.gz\n')
print(json.dumps({key: value for key, value in summary.items() if key not in ('static_result', 'unchanged_production_sha256')}, indent=2))
