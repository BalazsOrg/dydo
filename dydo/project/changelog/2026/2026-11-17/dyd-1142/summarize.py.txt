from pathlib import Path
import hashlib
import json
import subprocess
import xml.etree.ElementTree as ET

root = Path.cwd()
runs = Path('/tmp/dyd1142-runs')
owned = [
    'Services/SafeDeleteGitHub.cs',
    'Services/SafeDeleteGitHubTarget.cs',
    'Services/SafeDeleteService.cs',
    'DynaDocs.Tests/Integration/SafeDeleteGitHubTests.cs',
    'DynaDocs.Tests/Features/safe-delete.feature',
    'DynaDocs.Tests/Steps/SafeDeleteSteps.cs',
    'dydo/reference/dydo-commands.md',
    'Scaffold/dydo/reference/dydo-commands.md',
]
hashes = {name: hashlib.sha256((root / name).read_bytes()).hexdigest() for name in owned}
summary = {'base': subprocess.check_output(['git', 'rev-parse', 'HEAD'], text=True).strip(),
           'final_owned_source_sha256': hashes, 'runs': {}}
ns = {'t': 'http://microsoft.com/schemas/VisualStudio/TeamTest/2010'}
for receipt_path in sorted(runs.glob('*/receipt.json')):
    receipt = json.loads(receipt_path.read_text())
    if 'exit' not in receipt:
        continue
    label = receipt_path.parent.name
    row = {key: receipt.get(key) for key in ('head', 'exit', 'argv', 'start', 'end')}
    row['matches_final_owned_source'] = all(receipt['files'].get(name) == digest for name, digest in hashes.items())
    tests = []
    for trx in receipt_path.parent.glob('results/*.trx'):
        document = ET.parse(trx)
        tests.append({'path': str(trx), 'sha256': hashlib.sha256(trx.read_bytes()).hexdigest(),
                      'counters': document.find('.//t:Counters', ns).attrib,
                      'unpushed_gherkin': [node.attrib for node in document.findall('.//t:UnitTestResult', ns)
                                          if node.get('testName', '').startswith('Archive an unpushed')]})
    row['trx'] = tests
    summary['runs'][label] = row

red = json.loads((runs / 'red-production-reverted-02/receipt.json').read_text())
base_match = {}
for name in owned[:3]:
    original = subprocess.check_output(['git', 'show', 'd97c1b4fe9a7a843de8aceecfed5b9ad97ea2c9d:' + name])
    base_match[name] = hashlib.sha256(original).hexdigest() == red['files'][name]
assert all(base_match.values()), base_match
summary['second_red_production_matches_pinned_base'] = base_match
for label in ('discovery-restored', 'light-final', 'static-final', 'static-final-02', 'static-final-03'):
    assert summary['runs'][label]['matches_final_owned_source'], label

discovery = (runs / 'discovery-restored/raw.log').read_text()
summary['discovery'] = {'total': sum(line.startswith('    ') for line in discovery.splitlines()),
                        'new_gherkin': discovery.count('Archive an unpushed worktree'),
                        'api_cli_cases': discovery.count('SafeDeleteGitHubTests.'),
                        'command_doc_cases': discovery.count('CommandDocConsistencyTests.'),
                        'archive_registry_cases': discovery.count('Archive_IsRegisteredAndWorkerAllowed')}
static = json.loads((root / 'DynaDocs.Tests/coverage/results/adapters/dotnet-static.json').read_text())
summary['final_static_collectors'] = {name: {'status': row['status'], 'errors': row['errors'],
                                          'findings': row['findings']}
                                      for name, row in static['collectors'].items()}
assert all(row['status'] == 'pass' for row in static['collectors'].values())
summary['final_static_candidate'] = static['candidate']
(runs / 'summary.json').write_text(json.dumps(summary, indent=2) + '\n')
print(json.dumps({'discovery': summary['discovery'], 'second_red_matches_base': base_match,
                  'runs': {label: {'exit': row['exit'], 'matches_final_source': row['matches_final_owned_source'],
                                    'tests': [test['counters'] for test in row['trx']]}
                           for label, row in summary['runs'].items()}}, indent=2))
