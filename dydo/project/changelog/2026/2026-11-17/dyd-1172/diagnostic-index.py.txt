"""Index finished receipts after their writers have exited; no self hash."""
import hashlib
import gzip
import json
import sys
from pathlib import Path

evidence = Path(__file__).resolve().parent
index = evidence / "diagnostic-artifacts.json"
if "--pack" in sys.argv[1:]:
    reports = list(evidence.glob("diagnostic-*/adapters/*.json"))
    reports.append(evidence / "diagnostic-writer-forced/results/diagnostic.trx")
    for path in reports:
        if path.exists():
            raw = path.read_bytes()
            packed = Path(str(path) + ".gz")
            with packed.open("xb") as stream:
                stream.write(gzip.compress(raw, mtime=0))
            assert gzip.decompress(packed.read_bytes()) == raw
            path.unlink()
files = [p for p in evidence.glob("diagnostic-*") if p.is_file()]
for directory in evidence.glob("diagnostic-*"):
    if directory.is_dir():
        files.extend(p for p in directory.rglob("*") if p.is_file())
rows = {str(path.relative_to(evidence)): {"bytes": path.stat().st_size,
        "sha256": hashlib.sha256(path.read_bytes()).hexdigest()}
        for path in sorted(files) if path != index}
index.write_text(json.dumps(rows, indent=2) + "\n")
print(f"Indexed {len(rows)} finished artifacts; index excludes its own hash.")
