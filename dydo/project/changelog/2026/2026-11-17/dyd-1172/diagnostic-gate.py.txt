"""Use the existing capped runner; retain raw facade artifacts before another gate."""
import json
from importlib.machinery import SourceFileLoader
import importlib.util
from pathlib import Path
import shutil
import subprocess
import sys
import tarfile

EVIDENCE = Path(__file__).resolve().parent
ROOT = EVIDENCE.parents[5]
RESULTS = ROOT / "DynaDocs.Tests/coverage/results"
name, level, *command = sys.argv[1:]
before = set(RESULTS.glob("run-*")) | set((RESULTS / "assurance").glob("run-*"))
# The facade fingerprints all nonignored inputs. Its live streams must stay in
# ignored artifacts until collection ends, or output itself changes the inputs.
loader = SourceFileLoader("historical_runner", str(EVIDENCE / "run-proof.py.txt"))
spec = importlib.util.spec_from_loader(loader.name, loader)
runner = importlib.util.module_from_spec(spec)
sys.dont_write_bytecode = True
spec.loader.exec_module(runner)
runner.EVIDENCE = ROOT / "artifacts/dyd-1172-diagnostic"
runner.EVIDENCE.mkdir(parents=True, exist_ok=True)
sys.argv = [str(EVIDENCE / "run-proof.py.txt"), name, level, *command]
status = runner.main()
after = set(RESULTS.glob("run-*")) | set((RESULTS / "assurance").glob("run-*"))
paths = sorted(after - before)
destination = EVIDENCE / name
shutil.copytree(runner.EVIDENCE / name, destination)
if (RESULTS / "adapters").exists():
    shutil.copytree(RESULTS / "adapters", destination / "adapters")
with tarfile.open(destination / "facade-raw.tar.gz", "w:gz") as archive:
    for path in paths:
        archive.add(path, arcname=path.relative_to(ROOT))
(destination / "facade-artifacts.json").write_text(json.dumps({"exit": status,
    "retained": [str(p.relative_to(ROOT)) for p in paths]}, indent=2) + "\n")
raise SystemExit(status)
