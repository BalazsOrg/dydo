"""Retain completed sync evidence without changing executable repository inputs."""
import gzip
import hashlib
import json
from pathlib import Path
import re
import shutil
import subprocess
import tarfile
import xml.etree.ElementTree as ET

ROOT=Path.cwd()
LIVE=ROOT/'artifacts/dyd-1172-sync'
DEST=ROOT/'dydo/project/changelog/2026/2026-11-17/dyd-1172/sync-a5471145'
SOURCE='c6150e353d59cc3279f556f6ef8197c386359100'
TARGET='a5471145eb4b05b18f0829d1f7c5a00460ab5125'
MERGE='2078622fe2e22169898f06836929b2faf9e68f8c'

def git(*args):
    return subprocess.check_output(['git',*args],text=True).strip()

def sha(path):
    return hashlib.sha256(path.read_bytes()).hexdigest()

assert git('rev-parse','HEAD')==MERGE
assert not git('status','--porcelain=v1')
assert git('show','-s','--format=%P','HEAD').split()==[SOURCE,TARGET]
assert git('diff',SOURCE,'HEAD','--','*.cs')==''
assert git('diff',SOURCE,MERGE)==git('diff','ed2c5f06',TARGET)
required=['sync-prerequisites-final','sync-discovery','sync-safe-discovery','sync-selection-final',
          'sync-light','sync-static-dotnet','sync-coverage','sync-static-python','sync-docs']
receipts={p.parent.name:json.loads(p.read_text())['exit'] for p in LIVE.glob('sync-*/exit.json')}
assert all(name in receipts for name in required),receipts
for name in required:
    identity=json.loads((LIVE/name/'identity.json').read_text())
    assert identity['head']==MERGE and identity['afterHead']==MERGE
    assert identity['status']==identity['afterStatus']==''
    assert identity['csSha256']==identity['afterCsSha256']
before=json.loads((LIVE/'premerge.json').read_text())['sourceCsSha256']
assert all(sha(ROOT/path)==digest for path,digest in before.items())
selected=json.loads((LIVE/'selection.json').read_text())
owned_worktrees=sorted({match for path in LIVE.glob('sync-*/stdout.txt')
                       for match in re.findall(r'^\s*Worktree: (.+)$',path.read_text(),re.M)})
registered=git('worktree','list','--porcelain')
assert all(not Path(path).exists() and ('worktree '+path+'\n') not in registered+'\n'
           for path in owned_worktrees)
trx=ET.parse(LIVE/'sync-light/results/light.trx').find('.//{*}Counters').attrib
assert trx['passed']=='335' and trx['failed']=='0'
reports={}
for name,adapter in [('sync-static-dotnet','dotnet-static'),('sync-coverage','dotnet-coverage'),('sync-static-python','python-static')]:
    report=json.loads(gzip.decompress((LIVE/name/'adapters'/(adapter+'.json.gz')).read_bytes()))
    assert report['candidate']['commit']==MERGE and not report['candidate']['dirty']
    reports[name]={k:report.get(k) for k in ['candidate','status','exitCode','findings','errors']}
coverage=json.loads(gzip.decompress((LIVE/'sync-coverage/adapters/dotnet-coverage.json.gz').read_bytes()))
suite=coverage['collectors']['csharp-coverage']
suite_summaries=[]
with tarfile.open(LIVE/'sync-coverage/facade-raw.tar.gz','r:gz') as archive:
    for member in archive:
        if member.isfile() and member.name.endswith(('.stdout','.log')):
            text=archive.extractfile(member).read().decode(errors='replace')
            suite_summaries.extend({'path':member.name,'line':line} for line in text.splitlines()
                                   if 'Passed!' in line or ('Failed!' in line and 'Total:' in line)
                                   or re.match(r'^\s*(Total tests|Passed|Failed|Skipped|Total time):',line))
DEST.mkdir(exist_ok=False)
for path in LIVE.iterdir():
    if path.is_dir():
        shutil.copytree(path,DEST/path.name,ignore=shutil.ignore_patterns('adapters'))
    else:
        shutil.copy2(path,DEST/path.name)
for name,adapter in [('sync-static-dotnet','dotnet-static'),('sync-coverage','dotnet-coverage'),('sync-static-python','python-static')]:
    (DEST/name/'adapters').mkdir()
    shutil.copy2(LIVE/name/'adapters'/(adapter+'.json.gz'),DEST/name/'adapters')
for filename in ['capture.py','prerequisites.sh','selection.py','docs.py','summarize.py','finalize.py']:
    path=Path('/tmp')/('dyd1172-sync-'+filename)
    shutil.copy2(path,DEST/(filename+'.txt'))
diagnoses={
 'sync-prerequisites':'exit 1: sandbox denied user systemd scope bus; no probe child ran. Host-capped successor used.',
 'sync-prerequisites-host':'exit 1: probe requested nonexistent global.json after toolchain checks; remove that probe-only path.',
 'sync-prerequisites-corrected':'exit 2: capabilities does not accept repeated --stack flags; probe syntax fault.',
 'sync-prerequisites-valid':'exit 2: capabilities accepts no selector arguments at all; no-argument successor passed.',
 'sync-selection':'exit 1: probe counted only DynaDocs.* display names (253), excluding 82 selected Gherkin cases. Count all native indented names: 335. No selection changed.',
}
summary={
 'role':'code-writer/model: GPT-6.1 sol high', 'source':SOURCE,'pinnedTarget':TARGET,
 'merge':MERGE,'parents':[SOURCE,TARGET],'testedTree':git('rev-parse','HEAD^{tree}'),
 'automaticTargetDiffMatchesOriginalTargetDiff':True, 'allCsByteIdenticalToSource':True,
 'sourceCsSha256':before,'gateExits':receipts,'requiredGates':required,
 'requiredGatesPassed':all(receipts[name]==0 for name in required),
 'discovery':{k:v for k,v in selected.items() if k!='csSha256'},
 'lightCounters':trx,'assurance':reports,
 'coverageSuiteFacts':{k:suite['facts'].get(k) for k in ['child_exit','raw']},
 'coverageNativeSuiteSummaries':suite_summaries,
 'ownedWorktreeCleanup':{path:{'absent':True,'unregistered':True} for path in owned_worktrees},
 'hostGateLockSha256':sha(Path('/var/tmp/dydo-gate-lock.sh')),
 'probeFailureDiagnosis':diagnoses,
 'limits':[
  'Pre-review local sync only; no source PASS, PR, push or remote merge.',
  'Ticket coverage selection unavailable; supported whole-dotnet coverage producer and consumer used. Its child_exit=0 supplies the whole-suite verdict; no duplicate whole-suite run.',
  'No Heavy trigger. Mutation remains reviewer-owned UNAVAILABLE, Pending DYD-103; never claimed passing.',
  'Hosted Windows/Linux Actions and fresh Claude whole-head review remain captain-owned.',
  'Target pin supplied from 2026-11-17 remote read; exact local FETCH_HEAD verified. Sandbox remote read failed DNS; no retarget.',
  'Gates ran on the clean merge tree. Following hop adds only this inert evidence directory; C# hashes map that hop to the tested merge.',
 ],
 'memory':'Host-wide /var/tmp/dydo-gate-lock.sh outside env/systemd scope; probes 4G/120s, gates 8G/3600s, swap disabled, OOMPolicy=continue; no cap increase.',
 'mergeCommand':"git merge --no-ff "+TARGET+" -m 'DYD-1172 merge: sync pinned feature tip before review'",
 'mergeExit':0,'mergeReceipt':"Merge made by the 'ort' strategy.",
 'mergeMetadataAccess':'Initial sandbox merge failed creating ORIG_HEAD.lock in shared .git; authorized escalation succeeded with no conflict or manual resolution.',
}
(DEST/'receipt.json').write_text(json.dumps(summary,indent=2)+'\n')
lines=[f'DYD-1172 local pre-review sync: {MERGE}',
       f'Parents: {SOURCE} {TARGET}',f'Tested tree: {summary["testedTree"]}',
       'All C# files and the complete refusal-output assertion are byte-identical to source.',
       'Required gates: '+', '.join(f'{name} exit {receipts[name]}' for name in required),
       'SafeDelete Light: 335 passed, 0 failed, 0 skipped.',
       f'Whole-.NET suite verdict: coverage producer child_exit={suite["facts"]["child_exit"]}; native totals retained in coverage raw archive.',
       'Early failed probes remain failed; see receipt.json for each exact diagnosis.',
       *summary['limits'],'code-writer/model: GPT-6.1 sol high']
(DEST/'report.txt').write_text('\n'.join(lines)+'\n')
artifacts={str(path.relative_to(DEST)):{'bytes':path.stat().st_size,'sha256':sha(path)}
           for path in sorted(DEST.rglob('*')) if path.is_file()}
for path in DEST.rglob('facade-raw.tar.gz'):
    with tarfile.open(path,'r:gz') as archive:
        for member in archive:
            if member.isfile():
                stream=archive.extractfile(member)
                while stream.read(1024*1024):
                    pass
(DEST/'artifacts.json').write_text(json.dumps(artifacts,indent=2)+'\n')
print(json.dumps({'evidence':str(DEST),'merge':MERGE,'gateExits':receipts,'files':len(artifacts)},indent=2))
