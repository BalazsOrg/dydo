"""Bind the discovered native selections and candidate source bytes."""
import hashlib
import json
from pathlib import Path
import subprocess

EVIDENCE = Path(__file__).resolve().parent
ROOT = EVIDENCE.parents[5]
output = (EVIDENCE / "diagnostic-discovery/stdout.txt").read_text()
names = [line.strip() for line in output.splitlines() if line.startswith("    ")]
safe_output = (EVIDENCE / "diagnostic-safe-discovery/stdout.txt").read_text()
safe = [line.strip() for line in safe_output.splitlines() if line.startswith("    ")]
focused = [name for name in names if "SafeDeleteInputsTests.ChangedCommonInput_RetainsTargetEvenWhenStatusIsUnchanged" in name]
assert len(safe) == 335, len(safe)
assert len(focused) == 5, len(focused)
assert len(names) > len(safe)
tracked = subprocess.check_output(["git", "ls-files", "-z"], cwd=ROOT).decode().split("\0")
sources = {name: hashlib.sha256((ROOT / name).read_bytes()).hexdigest()
           for name in tracked if name and ((ROOT / name).suffix in {".cs", ".csproj", ".props", ".targets"}
           or name.startswith("DynaDocs.Tests/coverage/") or name in {"DynaDocs.sln", "global.json", ".config/dotnet-tools.json"})}
receipt = {"sourceHead": subprocess.check_output(["git", "rev-parse", "HEAD"], cwd=ROOT, text=True).strip(),
           "nativeDiscovered": len(names), "safeDeleteDiscovered": len(safe), "focusedDiscovered": focused,
           "safeDeleteSelection": "FullyQualifiedName~SafeDelete", "coverageSelection": "supported whole-dotnet route",
           "sourceSha256": sources}
(EVIDENCE / "diagnostic-preflight.json").write_text(json.dumps(receipt, indent=2) + "\n")
print(json.dumps({k: v for k, v in receipt.items() if k != "sourceSha256"}, indent=2))
