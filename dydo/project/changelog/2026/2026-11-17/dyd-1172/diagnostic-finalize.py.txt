"""Check restoration, source identity, gate verdicts and owned snapshot cleanup."""
import hashlib
import gzip
import json
from pathlib import Path
import subprocess
import tarfile
import xml.etree.ElementTree as ET

EVIDENCE = Path(__file__).resolve().parent
ROOT = EVIDENCE.parents[5]
source = "DynaDocs.Tests/Integration/SafeDeleteInputsTests.cs"
sha = lambda p: hashlib.sha256(p.read_bytes()).hexdigest()
git = lambda *args: subprocess.check_output(["git", *args], cwd=ROOT, text=True)
preflight = json.loads((EVIDENCE / "diagnostic-preflight.json").read_text())
for name, expected in preflight["sourceSha256"].items():
    assert sha(ROOT / name) == expected, name
changed = git("diff", "--name-only", "HEAD").splitlines()
prefix = "dydo/project/changelog/2026/2026-11-17/dyd-1172/"
assert [name for name in changed if not name.startswith(prefix)] == [source], changed
assert git("diff", "--diff-filter=MDR", "--name-only", "HEAD", "--", prefix) == ""
ns = {"t": "http://microsoft.com/schemas/VisualStudio/TeamTest/2010"}
for relative, expected in [("diagnostic-writer-restored/results/restored.trx", 5),
                           ("diagnostic-light/results/light.trx", 335)]:
    counters = ET.parse(EVIDENCE / relative).find(".//t:Counters", ns).attrib
    assert (int(counters["total"]), int(counters["passed"]), int(counters["failed"])) == (expected, expected, 0)
registered = git("worktree", "list", "--porcelain")
snapshots = []
for name in ["diagnostic-discovery", "diagnostic-safe-discovery", "diagnostic-writer-forced",
             "diagnostic-writer-restored", "diagnostic-light", "diagnostic-static", "diagnostic-coverage"]:
    for line in (EVIDENCE / name / "stdout.txt").read_text().splitlines():
        if line.startswith("  Worktree: "):
            path = line.split(": ", 1)[1]
            assert not Path(path).exists() and f"worktree {path}\n" not in registered, path
            snapshots.append(path)
fixture = (EVIDENCE / "diagnostic-writer-fixture-output.txt").read_text()
scratch = fixture.split("Space ", 1)[1].split("/main/.git:", 1)[0]
assert not Path(scratch).exists(), scratch
gates = {}
for name in ["diagnostic-light", "diagnostic-static", "diagnostic-static-stable",
             "diagnostic-static-base", "diagnostic-coverage"]:
    status = json.loads((EVIDENCE / name / "exit.json").read_text())["exit"]
    gates[name] = status
# Keep the initial failed counting helper as failed; its corrected successor is separate.
assert json.loads((EVIDENCE / "diagnostic-selection/exit.json").read_text())["exit"] == 1
assert json.loads((EVIDENCE / "diagnostic-selection-corrected/exit.json").read_text())["exit"] == 0
def report(relative):
    path = EVIDENCE / relative
    return json.loads(path.read_bytes() if path.exists() else gzip.decompress(Path(str(path) + ".gz").read_bytes()))

current = report("diagnostic-static-stable/adapters/dotnet-static.json")
base = report("diagnostic-static-base/adapters/dotnet-static.json")
assert current["findings"] == base["findings"]
for report in [current, base]:
    assert report["collectors"]["repository-inputs"]["status"] == "pass"
    assert not report["collectors"]["repository-inputs"]["errors"]
with tarfile.open(EVIDENCE / "diagnostic-native-facade-receipts.tar.gz", "w:gz") as archive:
    for path in sorted((ROOT / "DynaDocs.Tests/coverage/results").glob("run-*")):
        archive.add(path, arcname=path.relative_to(ROOT))
receipt = {"sourceHead": git("rev-parse", "HEAD").strip(), "sourceSha256": sha(ROOT / source),
           "historicalEvidenceUnchanged": True, "allPreflightSourceHashesMatch": True,
           "staticFindingsIdenticalOnPinnedSource": current["findings"],
           "gates": gates, "ownedSnapshotsAbsentAndUnregistered": snapshots,
           "probeScratchAbsent": scratch, "windowsReproduction": False,
           "mutation": "reviewer-owned UNAVAILABLE: Pending DYD-103"}
(EVIDENCE / "diagnostic-final.json").write_text(json.dumps(receipt, indent=2) + "\n")
artifacts = {str(path.relative_to(EVIDENCE)): {"bytes": path.stat().st_size, "sha256": sha(path)}
             for path in sorted(EVIDENCE.glob("diagnostic-*")) if path.is_file()
             and path.name != "diagnostic-artifacts.json"}
for directory in sorted(EVIDENCE.glob("diagnostic-*")):
    if directory.is_dir():
        for path in sorted(directory.rglob("*")):
            if path.is_file():
                artifacts[str(path.relative_to(EVIDENCE))] = {"bytes": path.stat().st_size, "sha256": sha(path)}
(EVIDENCE / "diagnostic-artifacts.json").write_text(json.dumps(artifacts, indent=2) + "\n")
print(json.dumps(receipt, indent=2))
