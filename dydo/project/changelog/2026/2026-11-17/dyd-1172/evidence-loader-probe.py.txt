"""Exercise the current .txt recipes without changing or running C# tests."""
import ast
from importlib.machinery import SourceFileLoader
import importlib.util
from pathlib import Path
import sys

sys.dont_write_bytecode = True
evidence = Path(__file__).resolve().parent
for name in ['diagnostic-probe.py.txt', 'probe-low-space.py.txt']:
    tree = ast.parse((evidence / name).read_text())
    paths = [node.value for node in ast.walk(tree) if isinstance(node, ast.Constant)]
    assert 'run-proof.py.txt' in paths and 'run-proof.py' not in paths, name
loader = SourceFileLoader('evidence_runner_probe', str(evidence / 'run-proof.py.txt'))
spec = importlib.util.spec_from_loader(loader.name, loader)
runner = importlib.util.module_from_spec(spec)
loader.exec_module(runner)
assert runner.ROOT == evidence.parents[5] and runner.EVIDENCE == evidence
assert callable(runner.main)
print('PASS: .txt SourceFileLoader imports the retained runner; diagnostic and low-space recipes reference the current runner path; no C# test run or edit.')
