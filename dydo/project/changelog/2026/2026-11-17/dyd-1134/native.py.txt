from pathlib import Path
import hashlib
import json
import os
import shutil
import subprocess
import tempfile

assembly = Path.cwd() / 'bin/Debug/net10.0/dydo.dll'
environment = {k: v for k, v in os.environ.items() if not k.startswith('GIT_')}
relative = 'logs/run-2026-11-16T21:25:15.log'
content = bytes([0, 255, 13, 10, 58, 128, 0]) + b'ignored timestamp payload\n'

def command(cwd, *args):
    result = subprocess.run(args, cwd=cwd, env=environment, text=True, capture_output=True)
    print('COMMAND', list(args), 'EXIT', result.returncode, flush=True)
    print(result.stdout, result.stderr, flush=True)
    return result

def run(cwd, *args):
    result = command(cwd, *args)
    assert result.returncode == 0
    return result.stdout.rstrip('\r\n')

with tempfile.TemporaryDirectory(prefix='dyd1134-native-') as folder:
    root = Path(folder)
    source, tree, destination, restored = [root / name for name in ('source', 'tree', 'destination', 'restored')]
    source.mkdir()
    git = lambda cwd, *args: run(cwd, 'git', *args)
    git(source, 'init', '-b', 'main')
    for key, value in [('user.name', 'DYD-1134 proof'), ('user.email', 'proof@example.invalid'), ('commit.gpgsign', 'false')]:
        git(source, 'config', key, value)
    (source / '.gitignore').write_bytes(b'logs/\n')
    (source / 'file.txt').write_bytes(b'base\n')
    git(source, 'add', '.')
    git(source, 'commit', '-m', 'seed')
    (source / 'dydo.json').write_text('{"archive":{"reserveBytes":0,"reservePercent":0}}')
    git(source, 'worktree', 'add', '-b', 'topic', str(tree))
    (tree / 'logs').mkdir()
    (tree / relative).write_bytes(content)
    status = git(tree, 'status', '--porcelain=v1', '--ignored', '--untracked-files=all')
    assert status == '!! ' + relative
    result = command(source, 'dotnet', str(assembly), 'archive', 'worktree', str(tree))
    if result.returncode != 0:
        assert tree.is_dir() and (tree / relative).read_bytes() == content
        assert git(source, 'rev-parse', '--verify', 'refs/heads/topic')
        print('REFUSAL WITNESS: exact colon filename/bytes and local ref retained', flush=True)
        raise AssertionError('Native Linux colon-name archive refused: ' + result.stderr + result.stdout)
    assert 'Independent proof passed: fresh object store; no alternates or inherited GIT_* variables' in result.stdout
    assert not tree.exists()
    assert git(source, 'for-each-ref', 'refs/heads/topic') == ''
    archives = source / '.git/dydo/safe-delete-archive'
    entry = next(path for path in archives.iterdir() if path.is_dir() and not path.name.startswith('.'))
    manifest = json.loads((entry / 'manifest.json').read_bytes())
    assert any(item['path'] == relative for item in manifest['files'])
    assert (entry / 'files' / relative).read_bytes() == content
    destination.mkdir()
    git(destination, 'init', '-b', 'main')
    (destination / 'dydo.json').write_text('{"archive":{"reserveBytes":0,"reservePercent":0}}')
    copied = destination / '.git/dydo/safe-delete-archive' / entry.name
    shutil.copytree(entry, copied)
    shutil.rmtree(source)
    assert not source.exists()
    assert not (destination / '.git/objects/info/alternates').exists()
    run(destination, 'dotnet', str(assembly), 'archive', 'restore', copied.name, '--path', str(restored), '--branch', 'recovered')
    assert git(restored, 'status', '--porcelain=v1', '--ignored', '--untracked-files=all') == status
    assert (restored / relative).read_bytes() == content
    assert [path.name for path in (restored / 'logs').iterdir()] == ['run-2026-11-16T21:25:15.log']
    assert copied.exists()
    print('NATIVE LINUX PROOF: archive/delete; original source repository removed; fresh destination object store; exact ignored colon filename and bytes recovered; sha256=' + hashlib.sha256(content).hexdigest(), flush=True)
assert not Path(folder).exists()
print('CLEANUP: owned temporary repositories removed', flush=True)
