from pathlib import Path
import hashlib
import json
import subprocess
import xml.etree.ElementTree as ET

root = Path('/tmp/dyd1134-runs')
evidence = Path('dydo/project/changelog/2026/2026-11-17/dyd-1134')
production = ['Services/SafeDeletePaths.cs']
tests = ['DynaDocs.Tests/Integration/SafeDeletePathsTests.cs', 'DynaDocs.Tests/coverage/dotnet-platforms.json']
paths = production + tests
digest = lambda data: hashlib.sha256(data).hexdigest()
runs = {}
for directory in sorted(root.iterdir()):
    receipt_path = directory / 'receipt.json'
    if not receipt_path.exists() or directory.name.startswith('evidence-'):
        continue
    receipt = json.loads(receipt_path.read_text())
    row = {key: receipt.get(key) for key in ('start', 'end', 'cwd', 'head', 'status', 'argv', 'exit')}
    row['files'] = {path: receipt['files'].get(path) for path in paths}
    row['raw_log'] = 'runs/' + directory.name + '/raw.log'
    row['trx'] = []
    for trx in directory.rglob('*.trx'):
        tree = ET.parse(trx)
        ns = {'t': 'http://microsoft.com/schemas/VisualStudio/TeamTest/2010'}
        counter = tree.find('.//t:Counters', ns)
        results = tree.findall('.//t:UnitTestResult', ns)
        row['trx'].append({'path': 'runs/' + str(trx.relative_to(root)), 'counters': counter.attrib,
                           'results': [{k: result.attrib.get(k) for k in ('testName', 'outcome')} for result in results]})
    log = (directory / 'raw.log').read_text()
    if directory.name.startswith('discovery'):
        discovered = [line.strip() for line in log.split('The following Tests are available:', 1)[1].splitlines() if line.startswith('    ')]
        row['discovered_count'] = len(discovered)
        row['feature_scenarios_discovered'] = any('Recover history and every worktree version independently' in line for line in discovered)
        assert discovered
    if directory.name == 'red-production-reversed':
        row['invalid_for_red_proof'] = 'Production restored before test process completed; passing result retained, never used as RED.'
    runs[directory.name] = row

base = '01c8d558dcda426128a866d1b537aa09c069387d'
original = {path: digest(subprocess.check_output(['git', 'show', base + ':' + path])) for path in production}
current = {path: digest(Path(path).read_bytes()) for path in paths}
assert all(runs['red-production-reversed-02']['files'][path] == original[path] for path in production)
assert all(runs[label]['exit'] == 1 for label in ('red-native-before-fix', 'red-before-fix', 'red-production-reversed-02'))
final_labels = ('discovery-restored-02', 'windows-selection-preflight', 'light-final', 'native-linux-final', 'static-final')
assert all(runs[label]['exit'] == 0 for label in ('prerequisites', 'discovery-red', 'green-initial', 'native-linux-initial', *final_labels))
assert all(runs[label]['files'][path] == current[path] for label in final_labels for path in paths)
assert all(runs['red-production-reversed-02']['files'][path] == current[path] for path in tests)
assert runs['discovery-restored-02']['feature_scenarios_discovered']
for label in ('red-before-fix', 'red-production-reversed-02'):
    assert len(runs[label]['trx']) == 1
    counters = runs[label]['trx'][0]['counters']
    assert counters['failed'] == '8' and counters['passed'] == '10' and counters['total'] == '18'
light = runs['light-final']['trx'][0]['counters']
assert light['failed'] == '0' and int(light['executed']) == int(light['total']) > 0
assert light['total'] == str(runs['discovery-restored-02']['discovered_count'])
native = (root / 'native-linux-final/raw.log').read_text()
assert 'original source repository removed' in native and 'CLEANUP: owned temporary repositories removed' in native
adapter = Path('DynaDocs.Tests/coverage/results/adapters/dotnet-static.json')
static = json.loads(adapter.read_text())
collectors = {name: {key: row.get(key) for key in ('status', 'errors', 'findings')} for name, row in static['collectors'].items()}
assert static['exitCode'] == 0 and static['measurementComplete'] is True and len(collectors) == 7
assert all(row['status'] == 'pass' and not row['errors'] and not row['findings'] for row in collectors.values())
recipes = {name: digest(Path('/tmp/dyd1134-' + name + '.py').read_bytes()) for name in ('run', 'native', 'selection', 'reverse', 'summarize')}
summary = {'base': base, 'batch_base': 'd97c1b4fe9a7a843de8aceecfed5b9ad97ea2c9d', 'runs': runs,
           'production_base_hashes': original, 'final_source_hashes': current,
           'production_only_reversal_matches_base': True, 'retained_tests_match_final_reversal': True,
           'final_source_bytes_match_discovery_selection_light_native_static': True,
           'static_adapter': {'candidate': static['candidate'], 'measurementComplete': static['measurementComplete'], 'collectors': collectors, 'sha256': digest(adapter.read_bytes())},
           'recipe_sha256': recipes, 'native_windows': 'OWED at whole-batch head; no simulated or Linux skip result counts as proof',
           'final_whole_stack_coverage_mutation_hosted': 'OWED at whole-batch head; mutation DYD-103 remains fail-closed UNAVAILABLE, never PASS'}
(evidence / 'summary.json').write_text(json.dumps(summary, indent=2) + '\n')
for name, row in runs.items():
    print(name, 'exit', row['exit'], 'discovered', row.get('discovered_count'), 'counters', [trx['counters'] for trx in row['trx']])
print('Seven static collectors:', json.dumps(collectors))
print('Production-only reversal equals pinned base; tests retained; final source bytes match discovery/selection/Light/native/static receipts.')
