from pathlib import Path
import hashlib
import json
import subprocess
import sys

target = Path('DynaDocs.Tests/coverage/tests/test_dotnet_platform.py')
original = target.read_bytes()
entries = (
    b"            'DynaDocs.Tests.Integration.SafeDeletePathsTests.UnsafePaths_RefuseTraversalAndForeignSeparators':\n"
    b"                'DynaDocs.Tests/Integration/SafeDeletePathsTests.cs',\n"
    b"            'DynaDocs.Tests.Integration.SafeDeletePathsTests.ColonPaths_AllowNativeNamesAndRefuseWindowsDrivesAndStreams':\n"
    b"                'DynaDocs.Tests/Integration/SafeDeletePathsTests.cs',\n"
)
assert original.count(entries) == 1
reversed_bytes = original.replace(entries, b'', 1)
assert reversed_bytes == subprocess.check_output(['git', 'show', '50a27f632c4d9d59707c305abfd18b3ad3fb25e2:' + str(target)])
try:
    target.write_bytes(reversed_bytes)
    result = subprocess.run([
        sys.executable, '/tmp/dyd1134-platform-selection-fix-run.py', 'reverse-red', 'probe',
        '.static-gates/python/bin/python', '-m', 'unittest', 'discover',
        '-s', 'DynaDocs.Tests/coverage/tests', '-p', 'test_dotnet_platform.py', '-v',
    ], check=False)
    assert result.returncode == 1, result.returncode
    log = Path('/tmp/dyd1134-platform-selection-fix-runs/reverse-red/raw.log').read_text()
    assert 'line 170' in log and "self.assertTrue(path.startswith('DynaDocs.Tests/Statechart/'))" in log
    assert 'Ran 32 tests' in log and 'FAILED (failures=1)' in log
finally:
    target.write_bytes(original)
    assert target.read_bytes() == original
    print(json.dumps({'restored_sha256': hashlib.sha256(target.read_bytes()).hexdigest(),
                      'restored_after_child_completion': True}), flush=True)
