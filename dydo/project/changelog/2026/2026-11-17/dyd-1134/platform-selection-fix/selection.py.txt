from pathlib import Path
import json
import subprocess
import sys

sys.path.insert(0, str(Path('DynaDocs.Tests/coverage').resolve()))
from dotnet_platform import discover_windows_tests, load_selection, native_platform, windows_filter

root = Path.cwd()
assert native_platform() == 'linux'
selection = load_selection(root, native_platform())
declared = discover_windows_tests(root, selection)
required = [
    'DynaDocs.Tests.Integration.SafeDeletePathsTests.UnsafePaths_RefuseTraversalAndForeignSeparators',
    'DynaDocs.Tests.Integration.SafeDeletePathsTests.ColonPaths_AllowNativeNamesAndRefuseWindowsDrivesAndStreams',
    'DynaDocs.Tests.Integration.SafeDeleteBoundaryTests.WindowsStreamMetadata_AcceptsDefaultStreamsAndRefusesAlternateStreams',
]
assert all(name in declared for name in required)
filter_value = windows_filter(selection)
print(json.dumps({'host': native_platform(), 'declared_windows_tests': declared, 'filter': filter_value}, indent=2), flush=True)
result = subprocess.run(['dotnet', 'test', 'DynaDocs.Tests/DynaDocs.Tests.csproj', '--no-restore', '--list-tests', '--filter', filter_value], text=True, capture_output=True)
print(result.stdout, result.stderr, flush=True)
assert result.returncode == 0
assert all(name in result.stdout for name in required)
print('SELECTION PREFLIGHT: new path theories and existing native ADS test included in actual hosted Windows filter; discovery on Linux only, NO native Windows execution claim', flush=True)
